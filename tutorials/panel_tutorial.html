Tutorial 1: Programming a Side Panel

This tutorial builds a dockable OpenCAD LISP side panel where users enter X/Y coordinates, add them to a table, pick a color and entity type (Lines or Circles), and draw them into the active drawing.

Step 1: Register the Panel Layout

Use oc-panel-register to define the panel ID, title, options (such as width, command, and optional style), and controls. OpenCAD LISP panels render in the dialog-like form style by default (pass (style . "palette") in the options alist if you want the compact Properties-grid style). Each button uses an explicit '((action . "(func)")) so clicking it evaluates a regular OpenCAD LISP defun directly.

(defun c:coordpanel ()
  (oc-panel-register "coord_panel" "Coordinate Drawer"
    '((width . 320) (command . "COORDPANEL"))

    (oc-info "lbl_status" "Enter X, Y and click Add Point.")

    (oc-row
      (oc-number "num_x" "X" 0.0)
      (oc-number "num_y" "Y" 0.0)
    )
    (oc-button "btn_add" "Add Point" '((action . "(coord-panel-add)")))

    (oc-table "tbl_pts" "Coordinates"
      '("X" "Y")
      '()
      '((allow_add . T) (allow_delete . T))
    )

    (oc-choice "cmb_color" "Color" '("Red" "Yellow" "Green" "Cyan" "Blue" "White") "Red")
    (oc-choice "cmb_type" "Entity Type" '("Lines" "Circles") "Lines")

    (oc-row
      (oc-button "btn_draw" "Draw Entities" '((action . "(coord-panel-draw)")))
      (oc-button "btn_clear" "Clear" '((action . "clear")))
    )
  )
  (oc-panel-toggle "coord_panel" T)
  (princ)
)
Why use '((action . "(coord-panel-add)"))? Side panel buttons with an S-expression action dispatch straight to OpenCAD LISP, allowing any normal defun to run without creating extra global c: commands.

Step 2: Add Coordinates to the Table

Read the current num_x, num_y, and tbl_pts values with oc-panel-get, append a new (x-str y-str) row, and push the updated table back using oc-panel-update.

(defun coord-panel-add (/ x y tbl new-row new-tbl)
  (setq x (oc-panel-get "coord_panel" "num_x"))
  (setq y (oc-panel-get "coord_panel" "num_y"))
  (setq tbl (oc-panel-get "coord_panel" "tbl_pts"))
  (if (not tbl) (setq tbl '()))

  (setq new-row (list (rtos (if x x 0.0) 2 4) (rtos (if y y 0.0) 2 4)))
  (setq new-tbl (append tbl (list new-row)))

  (oc-panel-update "coord_panel"
    (list
      (cons "tbl_pts" new-tbl)
      (cons "lbl_status" (strcat "Points in table: " (itoa (length new-tbl))))
    ))
  (princ)
)

Step 3: Draw Lines or Circles in the Selected Color

Convert each table row into a 3D point (x y 0.0), apply the chosen color via CECOLOR, and either:

  • Lines: Connect point 1 → point 2, point 2 → point 3, and so on.
  • Circles: Place a circle (radius 2.5) centered on every point in the table.
(defun coord-rows->pts (rows / pts)
  (setq pts '())
  (foreach r rows
    (if (>= (length r) 2)
      (setq pts (append pts (list (list (atof (nth 0 r)) (atof (nth 1 r)) 0.0))))
    )
  )
  pts
)

(defun coord-draw-entities (pts color-name entity-type / old-col i p1 p2)
  (setq old-col (getvar "CECOLOR"))
  (setvar "CECOLOR" (strcase color-name))

  (cond
    ((= entity-type "Lines")
     (setq i 0)
     (while (< (1+ i) (length pts))
       (setq p1 (nth i pts))
       (setq p2 (nth (1+ i) pts))
       (command "_.LINE" p1 p2 "")
       (setq i (1+ i))
     ))
    ((= entity-type "Circles")
     (foreach p1 pts
       (command "_.CIRCLE" p1 2.5)
     ))
  )

  (setvar "CECOLOR" old-col)
)

(defun coord-panel-draw (/ rows pts col etype)
  (setq rows  (oc-panel-get "coord_panel" "tbl_pts"))
  (setq col   (oc-panel-get "coord_panel" "cmb_color"))
  (setq etype (oc-panel-get "coord_panel" "cmb_type"))
  (setq pts   (coord-rows->pts (if rows rows '())))

  (cond
    ((null pts)
     (oc-panel-update "coord_panel" '((lbl_status . "Add at least one point first."))))
    ((and (= etype "Lines") (< (length pts) 2))
     (oc-panel-update "coord_panel" '((lbl_status . "Lines need at least 2 points."))))
    (T
     (coord-draw-entities pts (if col col "Red") (if etype etype "Lines"))
     (oc-panel-update "coord_panel"
       (list (cons "lbl_status" (strcat "Drew " etype " in " col ".")))))
  )
  (princ)
)
